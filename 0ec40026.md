## Code Template: Python Implementation

### Phase 1: Database Setup

```python
# database.py
import psycopg2
from datetime import datetime
import uuid

class OpportunityDB:
    def __init__(self, connection_string):
        self.conn = psycopg2.connect(connection_string)
        self.cur = self.conn.cursor()
    
    def create_opportunity(self, notice_id, title, agency):
        """Create new opportunity"""
        opportunity_id = str(uuid.uuid4())
        self.cur.execute("""
            INSERT INTO opportunities (id, notice_id, title, agency, created_at, analysis_status)
            VALUES (%s, %s, %s, %s, %s, 'UPLOADED')
        """, (opportunity_id, notice_id, title, agency, datetime.now()))
        self.conn.commit()
        return opportunity_id
    
    def store_document(self, opportunity_id, filename, extracted_text):
        """Store document and extracted text"""
        import hashlib
        doc_id = str(uuid.uuid4())
        text_hash = hashlib.sha256(extracted_text.encode()).hexdigest()
        
        self.cur.execute("""
            INSERT INTO documents 
            (id, opportunity_id, filename, extracted_text, text_hash, extraction_timestamp)
            VALUES (%s, %s, %s, %s, %s, %s)
        """, (doc_id, opportunity_id, filename, extracted_text, text_hash, datetime.now()))
        self.conn.commit()
        return doc_id
```

### Phase 2: PDF Extraction

```python
# extract_pdf.py
import pdfplumber

def extract_pdf_text(file_path):
    """Extract text from PDF"""
    text = ""
    with pdfplumber.open(file_path) as pdf:
        for page_num, page in enumerate(pdf.pages, 1):
            text += f"\n[PAGE {page_num}]\n"
            text += page.extract_text() + "\n"
    return text
```

### Phase 3: LLM Integration

```python
# classification.py
import anthropic
import json

class ClassificationPipeline:
    def __init__(self, api_key):
        self.client = anthropic.Anthropic(api_key=api_key)
    
    def classify_opportunity(self, document_list):
        """Run Prompt 1 to classify opportunity"""
        doc_summary = "\n\n".join([
            f"FILE: {doc['filename']}\nFIRST 300 CHARS: {doc['text'][:300]}..."
            for doc in document_list
        ])
        
        prompt = f"""ROLE: Government Solicitation Document Analyzer

TASK: Classify the structure of provided solicitation documents.

DOCUMENTS:
{doc_summary}

Respond in JSON format with classification, confidence, and extraction strategy."""
        
        message = self.client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1500,
            messages=[{"role": "user", "content": prompt}]
        )
        
        response_text = message.content[0].text
        json_start = response_text.find('{')
        json_end = response_text.rfind('}') + 1
        
        return json.loads(response_text[json_start:json_end])
```

## Frontend Component (React)

### DocumentViewer.jsx

```javascript
import React, { useState } from 'react';
import './DocumentViewer.css';

export function DocumentViewer({ opportunity }) {
  const [selectedDoc, setSelectedDoc] = useState(null);
  const [slideoutOpen, setSlideoutOpen] = useState(false);
  const [docContent, setDocContent] = useState(null);

  const handleDocumentClick = async (doc) => {
    setSelectedDoc(doc);
    
    // Fetch formatted content
    const response = await fetch(`/api/documents/${doc.id}/formatted`);
    const content = await response.json();
    
    setDocContent(content);
    setSlideoutOpen(true);
  };

  return (
    <div className="document-viewer">
      {/* Document List */}
      <div className="doc-list">
        <h3>Documents</h3>
        <div className="doc-items">
          {opportunity.documents.map(doc => (
            <div
              key={doc.id}
              className="doc-item"
              onClick={() => handleDocumentClick(doc)}
            >
              <div className="doc-name">{doc.filename}</div>
              <div className="doc-meta">
                <span>{doc.page_count} pages</span>
                <span className="section">{doc.likely_section}</span>
              </div>
            </div>
          ))}
        </div>
      </div>

      {/* Slideout Panel */}
      <div className={`slideout-panel ${slideoutOpen ? 'open' : ''}`}>
        <div className="slideout-header">
          <h3>{selectedDoc?.filename}</h3>
          <button onClick={() => setSlideoutOpen(false)}>✕</button>
        </div>

        {docContent && (
          <div className="slideout-content">
            <div className="summary-section">
              <h4>Summary</h4>
              <p>{docContent.summary}</p>
            </div>

            <div className="findings-section">
              <h4>Key Findings</h4>
              <ul>
                {docContent.highlights?.map((h, i) => (
                  <li key={i}>{h}</li>
                ))}
              </ul>
            </div>

            <div className="full-text-section">
              <h4>Full Content</h4>
              <div dangerouslySetInnerHTML={{ __html: docContent.markup }} />
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
```

## Database Schema

```sql
CREATE TABLE opportunities (
  id UUID PRIMARY KEY,
  notice_id VARCHAR(50) UNIQUE,
  title VARCHAR(500),
  agency VARCHAR(200),
  created_at TIMESTAMP,
  classification_type VARCHAR(20),
  analysis_status VARCHAR(20) DEFAULT 'UPLOADED'
);

CREATE TABLE documents (
  id UUID PRIMARY KEY,
  opportunity_id UUID REFERENCES opportunities(id),
  filename VARCHAR(300),
  extracted_text TEXT,
  text_hash VARCHAR(64),
  extraction_timestamp TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sections (
  id UUID PRIMARY KEY,
  document_id UUID REFERENCES documents(id),
  opportunity_id UUID REFERENCES opportunities(id),
  section_letter VARCHAR(1),
  section_title VARCHAR(300),
  start_position INT,
  end_position INT,
  confidence_level VARCHAR(10),
  content TEXT
);

CREATE TABLE requirements (
  id UUID PRIMARY KEY,
  opportunity_id UUID REFERENCES opportunities(id),
  requirement_id VARCHAR(50),
  requirement_text TEXT,
  category VARCHAR(50),
  compliance_level VARCHAR(20)
);

CREATE TABLE section_summaries (
  id UUID PRIMARY KEY,
  section_id UUID REFERENCES sections(id),
  summary_text TEXT,
  key_findings TEXT[],
  generated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```
